aki_mission_slot_1 = {
	slot = 1
	generic =  no
	ai = yes
	potential = {
		tag = AKI
		NOT = { map_setup = map_setup_random }
    }
	has_country_shield = yes

	AKI_a_god_who_loves_people = {
		icon = mission_build_up_to_force_limit
		position = 2
		required_missions = {
			AKI_loneliness_and_demise
		}
		trigger = {
			any_known_country = {
				calc_true_if = {
					amount = 2
					AND = {
						culture_group = japanese_g
						has_opinion = {
							who = AKI
							value = 75
						}
					}
				}
			}
		}
		effect = {
			add_country_modifier = {
				name = aki_love_for_humans
				duration = 360
			}
		}
	}

	AKI_autumns_peerless_beauty = {
		icon = mission_build_up_to_force_limit
		position = 8
		required_missions = {
			AKI_a_god_who_loves_people
		}
		trigger = {
			any_owned_province = {
				calc_true_if = {
					amount = 3
					AND = {
						has_terrain = enchanted_forest
						development = 10
					}
				}
			}
		}
		effect = {
			every_core_province = {
				limit = {
					has_terrain = enchanted_forest
				}
				add_province_modifier = {
					name = aki_beautiful_local_foliage
					duration = -1
				}
			}
		}
	}

	AKI_falling_leaves_of_madness = {
		icon = mission_build_up_to_force_limit
		position = 5
		required_missions = {
			AKI_autumns_peerless_beauty
		}
		trigger = {
			any_owned_province = {
				calc_true_if = {
					amount = 2
				}
				AND = {
					has_terrain = enchanted_forest
					has_building = ramparts
					has_fort_building_trigger = yes
				}
			}
		}
		effect = {
			every_core_province = {
				limit = {
					has_terrain = enchanted_forest
				}
				add_province_modifier = {
					name = aki_madness_from_falling_leaves
					duration = -1
				}
			}
		}
	}
}
