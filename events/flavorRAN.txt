			}
			set_country_flag = ran_loyalist_wave_3
			country_event = { id = flavor_ran.77 }
		}

			}
			set_country_flag = ran_loyalist_wave_2
			country_event = { id = flavor_ran.76 }
		}

				had_country_flag = { flag = ran_civil_war_started days = 548 }
				NOT = { has_country_flag = ran_loyalist_wave_1 }
			}
			set_country_flag = ran_loyalist_wave_1
			country_event = { id = flavor_ran.75 }
		}
		#Yukari's sabotage: after the first 3 months, about every 10 months, the same for every will level
		if = {
			limit = {
				had_country_flag = { flag = ran_civil_war_started days = 90 }
			}
			random = {
				chance = 10
				ran_yukari_sabotage = yes
			}
		}

			limit = { check_variable = { which = ran_own_will value = 11 } }

			}
		}

			limit = { check_variable = { which = ran_own_will value = 8 } }

				chance = 20

			}

	}
