namespace = flavor_tsk

#Event chain to spawn Tsuakasa
#The Fox' Advice
country_event = {
	id = flavor_tsk.100
	title = flavor_tsk.100.t
	desc = flavor_tsk.100.desc
	picture = BYAKUREN_eventPicture
	trigger = {
		ai = no
		NOT = { has_country_flag = th_had_tsukasa_chain_event }
		capital_scope = {
			continent = gensokyo_continent
		}
		OR = {
			overextension_percentage = 1.5
			AND = {
				overextension_percentage = 1
				num_of_loans = 5
			}
		}
	}
	fire_only_once = yes
	mean_time_to_happen = { months = 300 }
	immediate = { hidden_effect = { set_country_flag = th_had_tsukasa_chain_event } }
	option = {
		name = flavor_tsk.100.a #She might have an ingenius idea...
		custom_tooltip = flavor_tsk.100.a.tt
		add_adm_power = 75
		add_dip_power = 75
		add_mil_power = 75
		hidden_effect = {
			set_variable = {
				which = th_tsukasa_counter
				value = 1
			}
			set_country_flag = th_going_for_tsukasa_event_chain
		}
		ai_chance = {
			factor = 0
		}
	}
	option = {
		name = flavor_tsk.100.b #Begone wretched fox!
		custom_tooltip = flavor_tsk.100.b.tt
		ai_chance = {
			factor = 100
		}
	}
}

#Investment into the Fox Coin
country_event = {
	id = flavor_tsk.101
	title = flavor_tsk.101.t
	desc = flavor_tsk.101.desc
	picture = BYAKUREN_eventPicture
	trigger = {
		ai = no
		has_country_flag = th_going_for_tsukasa_event_chain
		NOT = { check_variable = { which = th_tsukasa_counter value = 6 } }
	}	
	mean_time_to_happen = { months = 12 }
	option = {
		name = flavor_tsk.101.a #
		add_years_of_income = -1
		add_inflation = 1
		hidden_effect = {
			random_list = {
				1 = { change_variable = { which = th_tsukasa_counter value = 1 } }
				1 = { change_variable = { which = th_tsukasa_counter value = 2 } }
				1 = { change_variable = { which = th_tsukasa_counter value = -1 } }
			}
		}
	}
	option = {
		name = flavor_tsk.end_this_madness # End this madness
		custom_tooltip = flavor_tsk.end_this_madness.tt
		add_stability = -2
		tooltip = { set_bankruptcy = root }
		hidden_effect = { clr_country_flag = th_going_for_tsukasa_event_chain }
	}
	after = {
		hidden_effect = {
			if = {
				limit = { NOT = { check_variable = { which = th_tsukasa_counter value = 0 } } }
				set_variable = { which = th_tsukasa_counter value = 0 }
			}	
		}
	}
}

country_event = {
	id = flavor_tsk.110
	title = flavor_tsk.110.t
	desc = flavor_tsk.110.desc
	picture = BYAKUREN_eventPicture
	trigger = {
		ai = no
		has_country_flag = th_going_for_tsukasa_event_chain
		check_variable = { which = th_tsukasa_counter value = 6 }
	}
	immediate = { hidden_effect = { clr_country_flag = th_going_for_tsukasa_event_chain } }
	mean_time_to_happen = { months = 12 }
	option = {
		name = flavor_tsk.110.a #
		define_ruler = {
			name = "Tsukasa"
			dynasty = "Kudamaki"
			culture = th_beast_youkai_culture
			adm = 4
			dip = 4
			mil = 4
			age = 15
			claim = 50
			female = yes
		}
		hidden_effect = {
			add_ruler_personality = immortal_personality
			set_ruler_flag = touhou_character_ruler
			set_ruler_flag = touhou_original_ruler
		}
		change_tag = TSK
		swap_non_generic_missions = yes
		on_change_tag_effect = yes
		swap_free_idea_group = yes
	}
	option = {
		name = flavor_tsk.110.b #
		custom_tooltip = flavor_tsk.110.b.tt
		add_stability = -4
		hidden_effect = {
			every_owned_province = {
				limit = { NOT = { is_core = TSK } }
				add_core = TSK
			}
		}
	}
}
