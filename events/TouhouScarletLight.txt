#Assume a Scarlet Tenet
country_event = {
	id = th_th_scarlet_christians_events.1
	title = th_th_scarlet_christians_events.1.t
	desc = th_th_scarlet_christians_events.1.d
	picture = OLDHAKUREI_eventPicture
	is_triggered_only = yes
	trigger = { religion = th_scarlet_christians }
	immediate = { hidden_effect = { set_country_flag = th_scarlet_christians_tenet_event_active } }
	after = {
		clr_country_flag = th_scarlet_christians_tenet_event_active
		clr_country_flag = th_scarlet_christians_changed_tenet_cd
		set_country_flag = th_scarlet_christians_changed_tenet_cd
	}
	
	option = {
		name = th_th_scarlet_christians_events.1.a
		trigger = { NOT = { th_scarlet_christians_has_tenet = { tenet = scarlet_vanity_tenet } } }
		th_scarlet_christians_take_tenet = { tenet = scarlet_vanity_tenet }
	}
	option = {
		name = th_th_scarlet_christians_events.1.b
		trigger = { NOT = { th_scarlet_christians_has_tenet = { tenet = elegant_charisma_tenet } } }
		th_scarlet_christians_take_tenet = { tenet = elegant_charisma_tenet }
	}
	option = {
		name = th_th_scarlet_christians_events.1.c
		trigger = { NOT = { th_scarlet_christians_has_tenet = { tenet = vampiric_ferocity_tenet } } }
		th_scarlet_christians_take_tenet = { tenet = vampiric_ferocity_tenet }
	}
}

#A Great Synod
country_event = {
	id = th_th_scarlet_christians_events.2
	title = th_th_scarlet_christians_events.2.t
	desc = th_th_scarlet_christians_events.2.d
	picture = OLDHAKUREI_eventPicture
	
	trigger = {
		uses_patriarch_authority = yes
		religion = th_scarlet_christians
		OR = {
			NOT = { has_country_flag = th_scarlet_christians_cd }
			had_country_flag = {
				flag = th_scarlet_christians_cd
				days = 1095
			}
		}
	}

	immediate = {
		clr_country_flag = th_scarlet_christians_cd
	}

	after = {
		set_country_flag = th_scarlet_christians_cd
	}
	
	mean_time_to_happen = {
		days = 1095
	}
	
	option = {
		name = orthodox_flavor.2.a
		add_patriarch_authority = 0.05
		add_estate_church_loyalty_effect = yes
	}
	option = {
		name = orthodox_flavor.2.b
		add_patriarch_authority = -0.05
		add_stability_or_adm_power = yes
		reduce_estate_church_loyalty_effect = yes
	}
}
