country_event = {
	id = flavor_voi.1
	title = flavor_voi.1.t
	desc = flavor_voi.1.desc
	picture = PATCHOULI_eventPicture
	fire_only_once = yes
	trigger = {
		has_idea = VOI_Elemental_Focus
		OR = {	
			tag = VOI
			was_tag = VOI
		}
		NOT = {
			has_country_modifier = th_voi_em_fire
			has_country_modifier = th_voi_em_water
			has_country_modifier = th_voi_em_wood
			has_country_modifier = th_voi_em_metal
			has_country_modifier = th_voi_em_earth
			has_country_modifier = th_voi_em_sun
			has_country_modifier = th_voi_em_moon
		}
	}
	
	immediate = {
		hidden_effect = { 
			trigger_switch = {
				on_trigger = has_country_modifier
				th_voi_em_fire = { remove_country_modifier = th_voi_em_fire }
				th_voi_em_water = { remove_country_modifier = th_voi_em_water }
				th_voi_em_wood = { remove_country_modifier = th_voi_em_wood }
				th_voi_em_metal = { remove_country_modifier = th_voi_em_metal }
				th_voi_em_earth = { remove_country_modifier = th_voi_em_earth }
				th_voi_em_sun = { remove_country_modifier = th_voi_em_sun }
				th_voi_em_moon = { remove_country_modifier = th_voi_em_moon }
			}
		}
	}
	option = {
		name = flavor_voi.1.a
		add_country_modifier = { 
			name = th_voi_em_fire
			duration = -1
		}
		ai_chance = { 
			factor = 1
		}
	
	}
	option = {
		name = flavor_voi.1.b
		add_country_modifier = { 
			name = th_voi_em_water
			duration = -1
		}
		ai_chance = { 
			factor = 1
		}
	
	}
	option = {

			duration = -1
		}
		ai_chance = { 
			factor = 1
		}
	
	}
	option = {
