###GENSOKYAN TERRAIN GROUPS###
# Every land province of Gensokyo carries an explicit custom terrain category
# (see the "Gensokyo Vanilla Terrain Replacer" section of map/terrain.txt), so
# the vanilla categories hills / highlands / mountain never match there anymore.
# These triggers restore the old three-step gradation, grouping the custom
# categories by defensive strength. The vanilla categories are kept in the lists
# so content also keeps working on provinces of the vanilla world.

# defence = 1, local_defensiveness ~ 0.1
th_has_hills_terrain_trigger = {
	OR = {
		has_terrain = hills
		has_terrain = genso_hills
		has_terrain = flower_hill
		has_terrain = ravine
		has_terrain = bedrock_highlands
		has_terrain = hellish_bumps
		has_terrain = lunar_hills
		has_terrain = celestial_hills
	}
}

# defence = 1, movement type mountains
th_has_highlands_terrain_trigger = {
	OR = {
		has_terrain = highlands
		has_terrain = genso_highlands
		has_terrain = autumn_hills
		has_terrain = autumn_highlands
		has_terrain = makai_mountain
		has_terrain = celestial_mountain
	}
}

# defence = 2
th_has_mountain_terrain_trigger = {
	OR = {
		has_terrain = mountain
		has_terrain = genso_mountains
		has_terrain = genbu_ravine
		has_terrain = lunar_mountains
		has_terrain = genso_peaks
	}
}

# Any of the three above
th_has_mountainous_terrain_trigger = {
	OR = {
		th_has_hills_terrain_trigger = yes
		th_has_highlands_terrain_trigger = yes
		th_has_mountain_terrain_trigger = yes
	}
}

# Mountainous ground plus the tengu towns built into it. tengu_urban carries no
# defence of its own, so it stays out of the three tiers above and only counts
# where the point is settlement rather than defensibility.
th_has_tengu_expansion_terrain_trigger = {
	OR = {
		th_has_mountainous_terrain_trigger = yes
		has_terrain = tengu_urban
	}
}

# Wet / hard-to-develop ground
th_has_marsh_terrain_trigger = {
	OR = {
		has_terrain = marsh
		has_terrain = genso_marsh
		has_terrain = swamp
	}
}

th_has_jungle_terrain_trigger = {
	OR = {
		has_terrain = jungle
		has_terrain = genso_jungle
		has_terrain = bamboo_forest
		has_terrain = bamboo_jungle
	}
}

th_has_ravine_terrain_trigger = {
	OR = {
		has_terrain = ravine
		has_terrain = genbu_ravine
	}
}

# Open, easy-to-develop ground
th_has_grasslands_terrain_trigger = {
	OR = {
		has_terrain = grasslands
		has_terrain = genso_grasslands
		has_terrain = nether_grassland
		has_terrain = celestial_grassland
	}
}

th_has_farmlands_terrain_trigger = {
	OR = {
		has_terrain = farmlands
		has_terrain = genso_farmlands
		has_terrain = fertile_fields
	}
}

# Everything the Swamp Development privilege cares about
th_has_swamp_developer_terrain_trigger = {
	OR = {
		th_has_marsh_terrain_trigger = yes
		th_has_jungle_terrain_trigger = yes
		th_has_ravine_terrain_trigger = yes
		th_has_grasslands_terrain_trigger = yes
		th_has_farmlands_terrain_trigger = yes
	}
}
