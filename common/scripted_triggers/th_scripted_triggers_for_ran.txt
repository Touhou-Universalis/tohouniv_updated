#RAN (Tsarstvo Lisy)

#flavor_ran.52 (The Amur Question): a country holding at least $size$ provinces in Manchuria, not RAN or its subject.
#Used to find the largest Manchurian owner when there is no Emperor of China.
ran_is_amur_rival = {
	NOT = { tag = ROOT }
	NOT = { is_subject_of = ROOT }
	num_of_owned_provinces_with = {
		value = $size$
		region = manchuria_region
	}
}

#ran_find_yukari_country: Yukari's country can take RAN as a vassal only if it is independent or RAN's own subject.
ran_can_be_ran_overlord = {
	NOT = { tag = ROOT }
	OR = {
		is_subject = no
		is_subject_of = ROOT
	}
}

#Country scope: can pay for Books for the Gap with power = adm / dip / mil (50 / 75 / 100 by ADM tech below 10 / 10+ / 20+).
ran_can_send_scholars = {
	OR = {
		AND = {
			NOT = { adm_tech = 10 }
			$power$_power = 50
		}
		AND = {
			adm_tech = 10
			NOT = { adm_tech = 20 }
			$power$_power = 75
		}
		AND = {
			adm_tech = 20
			$power$_power = 100
		}
	}
}
