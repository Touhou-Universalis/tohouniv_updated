custom_button = {
	name = th_scarlet_christians_select_new_tenet
	tooltip = th_scarlet_christians_select_new_tenet_desc
	potential = {
		religion = th_scarlet_christians
		NOT = { th_scarlet_christians_has_tenet = { tenet = elegant_charisma_tenet } }
		NOT = { th_scarlet_christians_has_tenet = { tenet = scarlet_vanity_tenet } }
		NOT = { th_scarlet_christians_has_tenet = { tenet = vampiric_ferocity_tenet } }
	}
	trigger = {
		custom_trigger_tooltip = {
			tooltip = th_scarlet_christians_has_not_changed_tenet_recently_tt
			OR = {
				NOT = { has_country_flag = th_scarlet_christians_changed_tenet_cd }
				had_country_flag = {
					flag = th_scarlet_christians_changed_tenet_cd
					days = 7300
				}
			}
		}
		hidden_trigger = {
			NOT = { has_country_flag = th_scarlet_christians_tenet_event_active }
		}
	}
	effect = {
		custom_tooltip = th_embrace_different_tenet_tt
		hidden_effect = {
			country_event = {
				id = th_th_scarlet_christians_events.1
			}
		}
	}
}
