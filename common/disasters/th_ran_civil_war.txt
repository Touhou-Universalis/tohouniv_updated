#RAN: the civil war against Yukari's Loyalists. Started by refusing Yukari in "Good Job, Ran" (flavor_ran.60):
#the option sets ran_refused_yukari and adds 100 progress (vanilla has no effect that starts a disaster directly).
#Win: 5 years passed and no loyalists left -> on_end flavor_ran.70 "The Last Spell Card".
#Lose: the loyalists enforce their demands (common/rebel_types/th_yukari_loyalists.txt) -> flavor_ran.73 "Back to Mayohiga".
#The barrage runs through the hidden monthly event flavor_ran.79, paced by Ran's hidden will.
th_ran_civil_war = {
	potential = {
		tag = RAN
		has_country_flag = ran_refused_yukari
		NOT = { has_country_flag = ran_civil_war_over }
	}

	can_start = {
		custom_trigger_tooltip = {
			tooltip = RAN_civil_war_refused_tt
			has_country_flag = ran_refused_yukari
		}
	}

	can_stop = {
		always = no
	}

	progress = {
		modifier = {
			factor = 100
			has_country_flag = ran_refused_yukari
		}
	}

	can_end = {
		custom_trigger_tooltip = {
			tooltip = RAN_civil_war_five_years_tt
			had_country_flag = { flag = ran_civil_war_started days = 1825 }
		}
		custom_trigger_tooltip = {
			tooltip = RAN_civil_war_no_loyalists_tt
			NOT = { has_spawned_rebels = th_yukari_loyalists }
		}
	}

	modifier = {
		global_unrest = 2
		land_morale = -0.1
	}

	on_start = flavor_ran.61
	on_end = flavor_ran.70

	on_monthly = {
		events = {
			flavor_ran.79
		}
	}
}
