#Accepts a culture unless it is already accepted or primary. The caller shows its own custom_tooltip.
ran_accept_culture = {
	if = {
		limit = {
			NOT = { primary_culture = $culture$ }
			NOT = { accepted_culture = $culture$ }
		}
		add_accepted_culture = $culture$
	}
}

#RAN_table_of_ranks swaps th_fox_autocracy_mechanic for th_fox_autocracy_ranks_mechanic. The two use different powers,
#so the gauges are read (in steps of 25) before the swap and refilled by flavor_ran.39 a day later.

		limit = { has_government_power = { mechanic_type = th_fox_autocracy_mechanic power_type = th_fox_$type$_power value = 125 } }
		set_variable = { which = ran_carry_$type$ value = 125 }

		limit = { has_government_power = { mechanic_type = th_fox_autocracy_mechanic power_type = th_fox_$type$_power value = 100 } }
		set_variable = { which = ran_carry_$type$ value = 100 }

		set_variable = { which = ran_carry_$type$ value = 75 }

		limit = { has_government_power = { mechanic_type = th_fox_autocracy_mechanic power_type = th_fox_$type$_power value = 50 } }
		set_variable = { which = ran_carry_$type$ value = 50 }

###THE FINALE: "GOOD JOB, RAN" AND THE CIVIL WAR (flavor_ran.60-79, common/disasters/th_ran_civil_war.txt)###

#Saves Yukari's country as event_target:ran_yukari_country: Mayohiga, Nether-Mayohiga, or a Gensokyo formed by Mayohiga
#(the mod already counts that one as Yukari's, decisions/MHG_decisions.txt). A country that is someone else's subject
#can't take RAN as a vassal, so it doesn't count; one that is OUR subject does (it is freed first, see ran_bow_to_yukari).
ran_find_yukari_country = {
	if = {
		limit = {
			exists = MHG
			MHG = { ran_can_be_ran_overlord = yes }
		}
		MHG = { save_event_target_as = ran_yukari_country }
	}
	else_if = {
		limit = {
			exists = NMG
			NMG = { ran_can_be_ran_overlord = yes }
		}
		NMG = { save_event_target_as = ran_yukari_country }
	}
	else_if = {
		limit = {
			exists = GNK
			GNK = {
				was_tag = MHG
				ran_can_be_ran_overlord = yes
			}
		}
		GNK = { save_event_target_as = ran_yukari_country }
	}
}

#RAN goes back to Yukari. Call ran_find_yukari_country first.
#With Yukari's country: freed if it is our subject, then RAN becomes its vassal.
#Without one: Yukari takes Ran's place as ruler, and RAN loses 100 prestige.
ran_bow_to_yukari = {
	if = {
		limit = { has_saved_event_target = ran_yukari_country }
		event_target:ran_yukari_country = {
			if = {
				limit = { is_subject_of = ROOT }
				grant_independence = yes
			}
			create_subject = {
				subject_type = vassal
				subject = ROOT
			}
		}
	}
	else = {
		define_ruler = {
			name = "Yukari"
			dynasty = "Yakumo"
			adm = 6
			dip = 6
			mil = 6
			age = 17
			female = yes
			culture = Youkai
			religion = ROOT
		}
		add_ruler_personality = immortal_personality
		set_ruler_flag = touhou_character_ruler
		set_ruler_flag = touhou_original_ruler
		set_ruler_flag = th_original_yukari
		add_prestige = -100
	}
}

#Country scope: one army of a reinforcement wave or a barrage spawn, in a gap province if there is one, otherwise anywhere
#but the capital. Provinces may be reused. Usage: ran_spawn_wave_loyalists = { size = 4 }
ran_spawn_wave_loyalists = {
	if = {
		limit = {
			any_owned_province = { has_province_flag = ran_gap_touched }
		}
		random_owned_province = {
			limit = { has_province_flag = ran_gap_touched }
			ran_spawn_loyalists_here = { size = $size$ }
		}
	}
	else = {
		random_owned_province = {
			limit = { is_capital = no }
			ran_spawn_loyalists_here = { size = $size$ }
		}
	}
}

#Country scope: saves event target ran_false_fox_home, where the False Fox appears.
#One of our provinces in the south (ruthenia_region or crimea_region) if we have any, otherwise any of ours but the capital.
ran_find_false_fox_home = {
	if = {
		limit = {
			any_owned_province = {
				is_capital = no
				OR = {
					region = ruthenia_region
					region = crimea_region
				}
			}
		}
		random_owned_province = {
			limit = {
				is_capital = no
				OR = {
					region = ruthenia_region
					region = crimea_region
				}
			}
			save_event_target_as = ran_false_fox_home
		}
	}
	else = {
		random_owned_province = {
			limit = { is_capital = no }
			save_event_target_as = ran_false_fox_home
		}
	}
}

#The False Fox rises (flavor_ran.94; user: pretender war, scaled to our size): an army of size $home$ in her province, then armies of
#size $extra$ in other owned provinces near it: 2 (1 when full = no), +1 at 50 owned provinces, +1 at 100. Never in the capital.
ran_false_fox_rise = {
	event_target:ran_false_fox_home = {
		spawn_rebels = {
			type = th_false_fox_rebels
			size = $home$
		}
		set_province_flag = ran_false_fox_risen
	}
	if = {
		limit = { always = $full$ }
		ran_false_fox_extra_rising = { size = $extra$ }
	}
	ran_false_fox_extra_rising = { size = $extra$ }
	if = {
		limit = { num_of_cities = 50 }
		ran_false_fox_extra_rising = { size = $extra$ }
	}
	if = {
		limit = { num_of_cities = 100 }
		ran_false_fox_extra_rising = { size = $extra$ }
	}
}
